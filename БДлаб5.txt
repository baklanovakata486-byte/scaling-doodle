-- создание таблиц и связей (CREATE TABLE)


/* Создание справочников */
CREATE TABLE departments (
  department_id INT PRIMARY KEY AUTO_INCREMENT,
  department_name VARCHAR(255) NOT NULL
);

CREATE TABLE building_types (
  building_type_id INT PRIMARY KEY AUTO_INCREMENT,
  building_type_name VARCHAR(255) NOT NULL
);

CREATE TABLE systems (
  system_id INT PRIMARY KEY AUTO_INCREMENT,
  system_name VARCHAR(255) NOT NULL
);

CREATE TABLE request_sources (
  source_id INT PRIMARY KEY AUTO_INCREMENT,
  source_name VARCHAR(255) NOT NULL
);

CREATE TABLE request_categories (
  category_id INT PRIMARY KEY AUTO_INCREMENT,
  category_name VARCHAR(255) NOT NULL
);

CREATE TABLE priorities (
  priority_id INT PRIMARY KEY AUTO_INCREMENT,
  priority_name VARCHAR(255) NOT NULL
);

CREATE TABLE criticality_levels (
  criticality_level_id INT PRIMARY KEY AUTO_INCREMENT,
  criticality_level_name VARCHAR(255) NOT NULL
);

CREATE TABLE work_statuses (
  status_id INT PRIMARY KEY AUTO_INCREMENT,
  status_name VARCHAR(255) NOT NULL
);

CREATE TABLE accident_types (
  accident_type_id INT PRIMARY KEY AUTO_INCREMENT,
  accident_type_name VARCHAR(255) NOT NULL,
  criticality_level_id INT,
  FOREIGN KEY (criticality_level_id) REFERENCES criticality_levels (criticality_level_id)
);

/* Создание основных сущностей */
CREATE TABLE requesters (
  requester_id INT PRIMARY KEY AUTO_INCREMENT,
  requester_full_name VARCHAR(255) NOT NULL,
  requester_phone VARCHAR(50) NOT NULL,
  requester_premises VARCHAR(100)
);

CREATE TABLE executors (
  executor_id INT PRIMARY KEY AUTO_INCREMENT,
  executor_full_name VARCHAR(255) NOT NULL,
  department_id INT,
  FOREIGN KEY (department_id) REFERENCES departments (department_id)
);

CREATE TABLE buildings (
  building_id INT PRIMARY KEY AUTO_INCREMENT,
  building_name VARCHAR(255) NOT NULL,
  building_type_id INT,
  building_address VARCHAR(255) NOT NULL,
  FOREIGN KEY (building_type_id) REFERENCES building_types (building_type_id)
);

CREATE TABLE equipment (
  equipment_id INT PRIMARY KEY AUTO_INCREMENT,
  equipment_name VARCHAR(255) NOT NULL,
  model VARCHAR(255),
  equipment_location VARCHAR(255) NOT NULL,
  building_id INT,
  FOREIGN KEY (building_id) REFERENCES buildings (building_id)
);

/* Реализация связи М:М через составной первичный ключ */
CREATE TABLE equipment_systems (
  equipment_id INT,
  system_id INT,
  PRIMARY KEY (equipment_id, system_id),
  FOREIGN KEY (equipment_id) REFERENCES equipment (equipment_id),
  FOREIGN KEY (system_id) REFERENCES systems (system_id)
);

/* Основные события */
CREATE TABLE requests (
  request_id INT PRIMARY KEY AUTO_INCREMENT,
  receipt_date DATE NOT NULL,
  source_id INT,
  requester_id INT,
  category_id INT,
  priority_id INT,
  accident_type_id INT,
  building_id INT,
  problem_location VARCHAR(255) NOT NULL,
  desired_due_date DATE,
  executor_id INT,
  assigned_at DATETIME NOT NULL,
  planned_due_date DATE NOT NULL,
  FOREIGN KEY (source_id) REFERENCES request_sources (source_id),
  FOREIGN KEY (requester_id) REFERENCES requesters (requester_id),
  FOREIGN KEY (category_id) REFERENCES request_categories (category_id),
  FOREIGN KEY (priority_id) REFERENCES priorities (priority_id),
  FOREIGN KEY (accident_type_id) REFERENCES accident_types (accident_type_id),
  FOREIGN KEY (building_id) REFERENCES buildings (building_id),
  FOREIGN KEY (executor_id) REFERENCES executors (executor_id)
);

CREATE TABLE work_executions (
  request_id INT,
  equipment_id INT,
  actual_completion_date DATE,
  work_description TEXT,
  work_cost DECIMAL(10, 2) NOT NULL DEFAULT 0,
  status_id INT,
  PRIMARY KEY (request_id, equipment_id),
  FOREIGN KEY (request_id) REFERENCES requests (request_id),
  FOREIGN KEY (equipment_id) REFERENCES equipment (equipment_id),
  FOREIGN KEY (status_id) REFERENCES work_statuses (status_id)
);

/* 3. Реализация бизнес-правила "Стоимость работ не может быть < 0"
   с помощью ТРИГГЕРА */
DELIMITER //
CREATE TRIGGER trg_work_cost_non_negative
BEFORE INSERT ON work_executions
FOR EACH ROW
BEGIN
  IF NEW.work_cost < 0 THEN
    SIGNAL SQLSTATE '45000'
      SET MESSAGE_TEXT = 'Стоимость работ не может быть отрицательной';
  END IF;
END//
DELIMITER ;

DELIMITER //
CREATE TRIGGER trg_work_cost_non_negative_upd
BEFORE UPDATE ON work_executions
FOR EACH ROW
BEGIN
  IF NEW.work_cost < 0 THEN
    SIGNAL SQLSTATE '45000'
      SET MESSAGE_TEXT = 'Стоимость работ не может быть отрицательной';
  END IF;
END//
DELIMITER ;

-- ЧАСТЬ 2. ЗАПРОСЫ

-- неявное соединение таблиц с фильтром (WHERE) и сортировкой (ORDER BY)


/* Запрос 1.1: Список аварийных заявок с контактами заявителей */
SELECT
    r.request_id AS Номер_заявки,
    r.receipt_date AS Дата_поступления,
    req.requester_full_name AS Заявитель,
    b.building_name AS Здание,
    c.category_name AS Категория,
    p.priority_name AS Приоритет,
    exec.executor_full_name AS Исполнитель,
    r.problem_location AS Место_проблемы
FROM requests r, requesters req, buildings b, request_categories c, priorities p, executors exec
WHERE r.requester_id = req.requester_id
  AND r.building_id = b.building_id
  AND r.category_id = c.category_id
  AND r.priority_id = p.priority_id
  AND r.executor_id = exec.executor_id
  AND c.category_name = 'Аварийная'
ORDER BY r.receipt_date DESC;

/* Запрос 1.2: Оборудование в бизнес-центрах */
SELECT
    e.equipment_name AS Оборудование,
    e.model AS Модель,
    e.equipment_location AS Расположение,
    b.building_name AS Здание
FROM equipment e, buildings b, building_types bt
WHERE e.building_id = b.building_id
  AND b.building_type_id = bt.building_type_id
  AND bt.building_type_name = 'Бизнес-центр'
ORDER BY b.building_name, e.equipment_location;

/* Запрос 1.3: Заявки из мобильного приложения */
SELECT
    r.request_id AS Номер_заявки,
    src.source_name AS Источник,
    exec.executor_full_name AS Исполнитель,
    d.department_name AS Подразделение,
    r.planned_due_date AS Планируемая_дата
FROM requests r, request_sources src, executors exec, departments d
WHERE r.source_id = src.source_id
  AND r.executor_id = exec.executor_id
  AND exec.department_id = d.department_id
  AND src.source_name = 'Мобильное приложение'
ORDER BY d.department_name;


-- явное соединение таблиц (JOIN)


/* Запрос 2.1: Отчет по выполненным работам (INNER JOIN) */
SELECT
    we.request_id AS Номер_заявки,
    e.equipment_name AS Оборудование,
    e.model AS Модель,
    we.actual_completion_date AS Дата_выполнения,
    we.work_cost AS Стоимость,
    ws.status_name AS Статус_работы,
    exec.executor_full_name AS Исполнитель,
    d.department_name AS Подразделение
FROM work_executions we
JOIN requests r ON we.request_id = r.request_id
JOIN equipment e ON we.equipment_id = e.equipment_id
JOIN work_statuses ws ON we.status_id = ws.status_id
JOIN executors exec ON r.executor_id = exec.executor_id
JOIN departments d ON exec.department_id = d.department_id
ORDER BY we.work_cost DESC;

/* Запрос 2.2: Сравнение плановых и фактических дат (LEFT JOIN) */
SELECT
    r.request_id AS Заявка,
    r.planned_due_date AS Плановая_дата,
    we.actual_completion_date AS Фактическая_дата,
    p.priority_name AS Приоритет
FROM requests r
LEFT JOIN work_executions we ON r.request_id = we.request_id
JOIN priorities p ON r.priority_id = p.priority_id
ORDER BY r.planned_due_date;

-- агрегирующие операции (SUM, AVG, COUNT, MAX, MIN)


/* Запрос 3.1: Общая финансовая статистика по ремонтам */
SELECT
    COUNT(*) AS Всего_работ,
    SUM(work_cost) AS Общая_сумма,
    AVG(work_cost) AS Средняя_стоимость,
    MAX(work_cost) AS Максимальная_стоимость,
    MIN(work_cost) AS Минимальная_стоимость
FROM work_executions;

/* Запрос 3.2: Статистика поступивших заявок */
SELECT
    COUNT(request_id) AS Всего_заявок,
    COUNT(DISTINCT requester_id) AS Уникальных_заявителей,
    COUNT(DISTINCT building_id) AS Задействовано_зданий
FROM requests;

/* Запрос 3.3: Средняя стоимость ремонта по типам зданий */
SELECT
    bt.building_type_name AS Тип_здания,
    AVG(we.work_cost) AS Средняя_стоимость_ремонта,
    MAX(we.work_cost) AS Самый_дорогой_ремонт
FROM work_executions we
JOIN requests r ON we.request_id = r.request_id
JOIN buildings b ON r.building_id = b.building_id
JOIN building_types bt ON b.building_type_id = bt.building_type_id
GROUP BY bt.building_type_name;


-- группировка GROUP BY и фильтрация HAVING


/* Запрос 4.1: Здания с высокими затратами на ремонт */
SELECT
    b.building_name AS Здание,
    COUNT(we.request_id) AS Количество_работ,
    SUM(we.work_cost) AS Суммарные_затраты
FROM work_executions we
JOIN requests r ON we.request_id = r.request_id
JOIN buildings b ON r.building_id = b.building_id
GROUP BY b.building_id, b.building_name
HAVING SUM(we.work_cost) > 5000
ORDER BY Суммарные_затраты DESC;

/* Запрос 4.2: Анализ аварий по уровням критичности */
SELECT
    cl.criticality_level_name AS Уровень_критичности,
    acc.accident_type_name AS Тип_аварии,
    COUNT(r.request_id) AS Количество_случаев
FROM requests r
JOIN accident_types acc ON r.accident_type_id = acc.accident_type_id
JOIN criticality_levels cl ON acc.criticality_level_id = cl.criticality_level_id
GROUP BY cl.criticality_level_name, acc.accident_type_name
HAVING COUNT(r.request_id) > 0
ORDER BY Количество_случаев DESC;


-- вложенные подзапросы (IN, ALL)


/* 1.1. Подзапрос с IN: заявки исполнителей из «Лифтовой службы» */
SELECT
    r.request_id,
    r.receipt_date,
    r.problem_location,
    ex.executor_full_name
FROM requests r
JOIN executors ex ON r.executor_id = ex.executor_id
WHERE r.executor_id IN (
    SELECT e.executor_id
    FROM executors e
    WHERE e.department_id = (
        SELECT d.department_id
        FROM departments d
        WHERE d.department_name = 'Лифтовая служба'
    )
)
ORDER BY r.receipt_date DESC;

/* 1.2. Подзапрос с ALL: работы дороже всех работ по заявке 400002 */
SELECT
    we.request_id,
    we.equipment_id,
    we.work_cost,
    we.work_description
FROM work_executions we
WHERE we.work_cost > ALL (
    SELECT we2.work_cost
    FROM work_executions we2
    WHERE we2.request_id = 400002
)
ORDER BY we.work_cost DESC;


-- функции управления потоком (IF, CASE)


/* 2.1. IF: выполнена ли работа в срок */
SELECT
    r.request_id,
    r.planned_due_date,
    we.actual_completion_date,
    IF(
        we.actual_completion_date IS NULL,
        'Не выполнена',
        IF(
            we.actual_completion_date <= r.planned_due_date,
            'В срок',
            'С просрочкой'
        )
    ) AS execution_status
FROM requests r
JOIN work_executions we ON r.request_id = we.request_id
ORDER BY r.request_id, we.equipment_id;

/* 2.2. CASE: категория стоимости работ */
SELECT
    we.request_id,
    we.equipment_id,
    we.work_cost,
    CASE
        WHEN we.work_cost < 3000 THEN 'Низкая'
        WHEN we.work_cost BETWEEN 3000 AND 8000 THEN 'Средняя'
        WHEN we.work_cost > 8000 THEN 'Высокая'
        ELSE 'Не определена'
    END AS cost_level
FROM work_executions we
ORDER BY we.work_cost DESC;

/* 2.3. CASE + IF: тип заявки и признак указания желаемого срока */
SELECT
    r.request_id,
    CASE
        WHEN r.accident_type_id IS NULL THEN 'Плановая'
        ELSE 'Аварийная'
    END AS request_kind,
    IF(r.desired_due_date IS NULL, 'Желаемый срок не указан', 'Желаемый срок указан') AS desired_due_flag,
    r.problem_location
FROM requests r
ORDER BY r.request_id;



-- представление (CREATE VIEW) и временное представление (WITH)


/* 3.1. CREATE VIEW: отчёт по выполненным работам */
CREATE OR REPLACE VIEW v_work_report AS
SELECT
    we.request_id,
    we.equipment_id,
    e.equipment_name,
    we.actual_completion_date,
    we.work_cost,
    ws.status_name
FROM work_executions we
JOIN equipment e ON we.equipment_id = e.equipment_id
JOIN work_statuses ws ON we.status_id = ws.status_id;

SELECT * FROM v_work_report ORDER BY work_cost DESC;

/* 3.2. WITH: суммарные затраты по зданиям */
WITH building_costs AS (
    SELECT
        b.building_id,
        b.building_name,
        COUNT(we.request_id) AS works_count,
        SUM(we.work_cost) AS total_cost
    FROM work_executions we
    JOIN requests r ON we.request_id = r.request_id
    JOIN buildings b ON r.building_id = b.building_id
    GROUP BY b.building_id, b.building_name
)
SELECT
    building_name,
    works_count,
    total_cost
FROM building_costs
WHERE total_cost > 5000
ORDER BY total_cost DESC;

/* 3.3. WITH: нагрузка и стоимость работ по исполнителям */
WITH executor_work AS (
    SELECT
        ex.executor_id,
        ex.executor_full_name,
        d.department_name,
        COUNT(we.request_id) AS works_count,
        SUM(we.work_cost) AS total_cost
    FROM executors ex
    JOIN departments d ON ex.department_id = d.department_id
    JOIN requests r ON ex.executor_id = r.executor_id
    JOIN work_executions we ON r.request_id = we.request_id
    GROUP BY ex.executor_id, ex.executor_full_name, d.department_name
)
SELECT *
FROM executor_work
ORDER BY total_cost DESC;
